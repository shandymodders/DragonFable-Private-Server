   // -----------------------------------------------------------------------
   // AREA ID
   // -----------------------------------------------------------------------
   // DragonFable can use multiple frames/rooms inside the same map SWF.
   // Include all semantic room indicators we can read so remote players from
   // another room are removed instead of leaking into the current area.
   _root.dfpsGetMapId = function()
   {
      var map = _root.game.stage.map;
      if(map == undefined)
      {
         return "";
      }

      var mapUrl = "";
      if(map._url != undefined)
      {
         mapUrl = String(map._url);
      }

      var mapFrame = 0;
      if(map._currentframe != undefined)
      {
         mapFrame = Number(map._currentframe);
      }

      var mapLabel = "";
      if(map._currentlabel != undefined)
      {
         mapLabel = String(map._currentlabel);
      }

      var placementFrame = "";
      if(_root.game.character != undefined && _root.game.character.placement != undefined && _root.game.character.placement.strFrame != undefined)
      {
         placementFrame = String(_root.game.character.placement.strFrame);
      }

      return mapUrl + "|room=" + String(mapFrame) + "|label=" + mapLabel + "|placement=" + placementFrame;
   };

   // Multiplayer is intentionally town/public-map only. Quest instances and
   // every active battle are private gameplay contexts: remote avatars must
   // never be created there and the browser bridge is asked to leave the room.
   _root.dfpsIsPrivateWorld = function()
   {
      if(_root.battle != undefined && _root.battle.isActive == true)
      {
         return true;
      }
      if(_root.quest != undefined && _root.town == undefined)
      {
         return true;
      }
      return false;
   };

   // Resolve the class SWF that is ACTUALLY loaded on the local avatar.
   // Temporary classes loaded through cf-classload.asp are not necessarily
   // persisted in the character database, so ptrObj/database class alone can
   // be stale. avatar._url points at the class SWF currently being rendered.
   _root.dfpsGetActiveClassFile = function(avatar,playerObj)
   {
      var fallback = "";
      if(playerObj != undefined && playerObj.strClassFileName != undefined)
      {
         fallback = String(playerObj.strClassFileName);
      }

      if(avatar == undefined || avatar._url == undefined)
      {
         return fallback;
      }

      var avatarUrl = String(avatar._url);
      var queryPos = avatarUrl.indexOf("?");
      if(queryPos >= 0)
      {
         avatarUrl = avatarUrl.substr(0,queryPos);
      }
      var hashPos = avatarUrl.indexOf("#");
      if(hashPos >= 0)
      {
         avatarUrl = avatarUrl.substr(0,hashPos);
      }

      var lowerUrl = avatarUrl.toLowerCase();
      var classPos = lowerUrl.lastIndexOf("classes/");
      if(classPos >= 0)
      {
         var remainder = avatarUrl.substr(classPos + 8);
         var genderSlash = remainder.indexOf("/");
         if(genderSlash >= 0 && genderSlash < remainder.length - 1)
         {
            remainder = remainder.substr(genderSlash + 1);
         }
         if(remainder != "")
         {
            return remainder;
         }
      }
      return fallback;
   };

   _root.dfpsGetActiveAnimationLabel = function(avatar)
   {
      if(avatar == undefined)
      {
         return "";
      }

      var label = "";
      if(avatar._currentlabel != undefined)
      {
         label = String(avatar._currentlabel);
      }
      else if(avatar.currentLabel != undefined)
      {
         label = String(avatar.currentLabel);
      }

      if(label == "undefined" || label == "null")
      {
         return "";
      }
      return label;
   };

   _root.dfpsFindRuntimeVisualItem = function(playerObj,primarySpot,alternateSpot)
   {
      if(playerObj == undefined)
      {
         return undefined;
      }

      var lists = new Array();
      lists.push(playerObj.arrTempInventory);
      lists.push(playerObj.items);

      var usedItem = undefined;
      for(var listIndex = 0; listIndex < lists.length; listIndex++)
      {
         var items = lists[listIndex];
         if(items == undefined)
         {
            continue;
         }

         for(var itemIndex = 0; itemIndex < items.length; itemIndex++)
         {
            var item = items[itemIndex];
            if(item == undefined)
            {
               continue;
            }

            var spot = String(item.strEquipSpot).toLowerCase();
            if(spot != String(primarySpot).toLowerCase() && spot != String(alternateSpot).toLowerCase())
            {
               continue;
            }

            var fileName = String(item.strFileName);
            if(fileName == "" || fileName == "undefined" || fileName == "null" || fileName.toLowerCase() == "none")
            {
               continue;
            }

            // "showed" is the cosmetic item actually rendered by DragonFable.
            // It can intentionally differ from the stat item marked as "used".
            if(item.showed == true)
            {
               return item;
            }
            if(usedItem == undefined && item.used == true)
            {
               usedItem = item;
            }
         }
      }

      return usedItem;
   };

   _root.dfpsDescribeRuntimeItem = function(item,visible)
   {
      if(item == undefined)
      {
         return {file:"",itemType:"",type:"",visible:0};
      }
      return {
         file:String(item.strFileName),
         itemType:String(item.strItemType),
         type:String(item.strType),
         visible:visible == true ? 1 : 0
      };
   };

   _root.dfpsGetRuntimeEquipment = function(playerObj,avatar)
   {
      var weapon = _root.dfpsFindRuntimeVisualItem(playerObj,"weapon","");
      var back = _root.dfpsFindRuntimeVisualItem(playerObj,"back","cape");
      var head = _root.dfpsFindRuntimeVisualItem(playerObj,"head","helm");

      var weaponVisible = weapon != undefined;
      var backVisible = back != undefined;
      var headVisible = head != undefined;

      return {
         weapon:_root.dfpsDescribeRuntimeItem(weapon,weaponVisible),
         back:_root.dfpsDescribeRuntimeItem(back,backVisible),
         head:_root.dfpsDescribeRuntimeItem(head,headVisible)
      };
   };

   _root.dfpsIsNumericAttribute = function(attrName)
   {
      if(attrName == "CharID" || attrName == "ClassID" || attrName == "BaseClassID" || attrName == "RaceID" || attrName == "ItemID" || attrName == "CharItemID")
      {
         return true;
      }
      if(attrName.substr(0,3) == "int" || attrName.substr(0,3) == "bit")
      {
         return true;
      }
      return false;
   };

   _root.dfpsNodeToObject = function(node)
   {
      var obj = new Object();
      if(node == undefined)
      {
         return obj;
      }

      for(var attrName in node.attributes)
      {
         var attrValue = node.attributes[attrName];
         if(_root.dfpsIsNumericAttribute(attrName))
         {
            obj[attrName] = Number(attrValue);
         }
         else
         {
            obj[attrName] = String(attrValue);
         }
      }
      return obj;
   };

   _root.dfpsIsVisualEquipSpot = function(spot)
   {
      spot = String(spot).toLowerCase();
      return spot == "weapon" || spot == "back" || spot == "cape" || spot == "head" || spot == "helm";
   };

   _root.dfpsParseAppearance = function(xmlText)
   {
      var appearanceXML = new XML();
      appearanceXML.ignoreWhite = true;
      appearanceXML.parseXML(String(xmlText));

      var charNode = appearanceXML.firstChild;
      if(charNode != undefined && charNode.nodeName != "character")
      {
         var rootChildren = charNode.childNodes;
         for(var rc = 0; rc < rootChildren.length; rc++)
         {
            if(rootChildren[rc].nodeName == "character")
            {
               charNode = rootChildren[rc];
               break;
            }
         }
      }

      if(charNode == undefined || charNode.nodeName != "character")
      {
         return undefined;
      }

      var obj = _root.dfpsNodeToObject(charNode);
      obj.items = new Array();
      obj.arrTempInventory = new Array();
      obj.bitNPC = false;
      obj.petDragon = false;

      var childNodes = charNode.childNodes;
      for(var i = 0; i < childNodes.length; i++)
      {
         if(childNodes[i].nodeName == "items")
         {
            var itemObj = _root.dfpsNodeToObject(childNodes[i]);
            if(!_root.dfpsIsVisualEquipSpot(itemObj.strEquipSpot))
            {
               continue;
            }

            // Keep the visual flag separately. DragonFable's normal equipment
            // pipeline must never be allowed to process remote Armor/Class data.
            itemObj.dfpsEquipped = Number(itemObj.bitEquipped) == 1;
            itemObj.bitEquipped = 0;
            itemObj.used = false;
            itemObj.showed = false;
            itemObj.slotted = false;
            obj.items.push(itemObj);
         }
      }

      return obj;
   };

   _root.dfpsAllocateUnitIndex = function()
   {
      var idx = _root.dfpsNextUnitIndex;
      if(_root.game.Units == undefined || _root.game.Units.arrUnits == undefined)
      {
         return -1;
      }

      while(_root.game.Units.arrUnits[idx] != undefined)
      {
         idx++;
      }
      _root.dfpsNextUnitIndex = idx + 1;
      return idx;
   };

   // -----------------------------------------------------------------------
   // REMOTE-SAFE SILENT LOADER
   // -----------------------------------------------------------------------
   // The stock Loader.loadClip() shows the global loading screen and increments
   // intLoadCount. Its onLoadError path does NOT decrement intLoadCount, so one
   // missing remote cape/helm can permanently freeze login or map loading.
   // Remote assets therefore never use game.mcConn.

   _root.dfpsFindRemoteEntryForClip = function(mc)
   {
      var cursor = mc;
      var depthGuard = 0;
      while(cursor != undefined && depthGuard < 24)
      {
         for(var remoteKey in _root.dfpsRemotePlayers)
         {
            var entry = _root.dfpsRemotePlayers[remoteKey];
            if(entry != undefined && entry.clip != undefined && cursor == entry.clip)
            {
               return entry;
            }
         }
         cursor = cursor._parent;
         depthGuard++;
      }
      return undefined;
   };

   _root.dfpsRemovePendingLoad = function(entry,info)
   {
      if(entry == undefined || entry.pendingLoads == undefined)
      {
         return;
      }
      for(var i = entry.pendingLoads.length - 1; i >= 0; i--)
      {
         if(entry.pendingLoads[i] == info)
         {
            entry.pendingLoads.splice(i,1);
            return;
         }
      }
   };

   _root.dfpsSilentLoadFinished = function(info,success,errorCode)
   {
      if(info == undefined || info.finished == true)
      {
         return;
      }
      info.finished = true;

      var entry = info.entry;
      if(entry != undefined)
      {
         _root.dfpsRemovePendingLoad(entry,info);
      }

      // Ignore callbacks belonging to an avatar that was destroyed/rebuilt.
      if(entry == undefined || Number(entry.loaderGeneration) != Number(info.loaderGeneration))
      {
         return;
      }

      // A previous cape/helm/weapon request may finish after the sender has
      // already changed equipment. Never let that stale asset become visible.
      if((info.kind == "weapon" || info.kind == "back" || info.kind == "helmet") && String(info.equipmentKey) != String(entry.equipmentKey))
      {
         return;
      }

      if(success == true)
      {
         if(info.kind == "character")
         {
            entry.appearanceLoading = false;
            entry.appearanceReady = true;
            entry.appearanceFailed = false;
            entry.appearanceFailureCount = 0;

            if(entry.clip != undefined)
            {
               entry.clip._visible = true;
            }

            if(entry.avatar != undefined && entry.avatar.ptrMC != undefined)
            {
               _root.dfpsApplyRemoteFacing(entry);
               _root.dfpsApplyRuntimeEquipment(entry,true);
               _root.dfpsApplyRemoteAnimation(entry,true);
            }
         }
      }
      else
      {
         if(info.kind == "character")
         {
            entry.appearanceLoading = false;
            entry.appearanceReady = false;
            entry.appearanceFailed = true;
            entry.appearanceFailureCount = Number(entry.appearanceFailureCount) + 1;
         }

         // Equipment failure is intentionally non-fatal. A missing remote cape,
         // helm or weapon must never block the local player's game.
         if(info.target != undefined && info.kind != "character")
         {
            info.target._visible = false;
         }
      }
   };

   _root.dfpsSilentLoadAbsolute = function(mcDestination,strUrl,strDescription,entry,kind)
   {
      if(mcDestination == undefined || entry == undefined)
      {
         return false;
      }

      strUrl = String(strUrl);
      if(strUrl == "" || strUrl.toLowerCase() == "none" || strUrl == "undefined" || strUrl == "null")
      {
         return false;
      }

      if(entry.pendingLoads == undefined)
      {
         entry.pendingLoads = new Array();
      }

      var info = new Object();
      info.entry = entry;
      info.target = mcDestination;
      info.url = strUrl;
      info.description = String(strDescription);
      info.kind = String(kind);
      info.startedAt = getTimer();
      info.finished = false;
      info.loaderGeneration = Number(entry.loaderGeneration);
      info.equipmentKey = String(entry.equipmentKey);

      var listener = new Object();
      listener.info = info;
      listener.onLoadInit = function(mcTarget)
      {
         _root.dfpsSilentLoadFinished(this.info,true,"");
      };
      listener.onLoadComplete = function(mcTarget)
      {
      };
      listener.onLoadError = function(mcTarget,errorCode,httpStatus)
      {
         _root.dfpsSilentLoadFinished(this.info,false,String(errorCode));
      };

      var mcl = new MovieClipLoader();
      info.loader = mcl;
      info.listener = listener;
      mcl.addListener(listener);
      entry.pendingLoads.push(info);

      mcDestination.isValid = true;
      var accepted = mcl.loadClip(strUrl,mcDestination);
      if(accepted == false)
      {
         _root.dfpsSilentLoadFinished(info,false,"LoadRejected");
         return false;
      }
      return true;
   };

   _root.dfpsSilentLoadFile = function(mcDestination,strTarget,strDescription,entry,kind)
   {
      strTarget = String(strTarget);
      if(strTarget == "" || strTarget.toLowerCase() == "none" || strTarget == "undefined" || strTarget == "null")
      {
         return false;
      }
      return _root.dfpsSilentLoadAbsolute(mcDestination,String(_root.core.path) + strTarget,strDescription,entry,kind);
   };

   _root.dfpsSilentLoadRemoteCharacter = function(entry)
   {
      if(entry == undefined || entry.avatar == undefined || entry.avatar.ptrObj == undefined || entry.clip == undefined)
      {
         return false;
      }

      var obj = entry.avatar.ptrObj;
      var gender = String(obj.strGender).toUpperCase();
      var classFile = String(obj.strClassFileName);
      if(gender == "" || classFile == "" || classFile == "undefined" || classFile == "null")
      {
         entry.appearanceLoading = false;
         entry.appearanceFailed = true;
         entry.appearanceFailureCount = Number(entry.appearanceFailureCount) + 1;
         return false;
      }

      var classPath = String(_root.core.path) + "classes/" + gender + "/" + classFile;
      entry.appearanceLoading = true;
      entry.appearanceReady = false;
      entry.clip._visible = false;
      return _root.dfpsSilentLoadAbsolute(entry.clip,classPath,"DFPS Remote Character",entry,"character");
   };

   _root.dfpsSilentLoaderWatchdog = function()
   {
      var now = getTimer();
      for(var remoteKey in _root.dfpsRemotePlayers)
      {
         var entry = _root.dfpsRemotePlayers[remoteKey];
         if(entry == undefined || entry.pendingLoads == undefined)
         {
            continue;
         }

         for(var i = entry.pendingLoads.length - 1; i >= 0; i--)
         {
            var info = entry.pendingLoads[i];
            if(info == undefined || info.finished == true)
            {
               entry.pendingLoads.splice(i,1);
               continue;
            }

            if(now - Number(info.startedAt) > _root.dfpsRemoteLoadTimeout)
            {
               if(info.loader != undefined && info.target != undefined)
               {
                  info.loader.unloadClip(info.target);
               }
               _root.dfpsSilentLoadFinished(info,false,"Timeout");
            }
         }
      }
   };

   // Intercept normal conn.loadFile only when the destination belongs to a
   // remote avatar. Local character/map/interface assets still use the original
   // DragonFable loader without any change.
   if(_root.conn != undefined && _root.conn.loadFile != undefined)
   {
      _root.dfpsOriginalConnLoadFile = _root.conn.loadFile;
      _root.conn.loadFile = function(mcDestination,strTarget,strDescription)
      {
         var remoteEntry = _root.dfpsFindRemoteEntryForClip(mcDestination);
         if(remoteEntry != undefined)
         {
            var kind = "equipment";
            var descLower = String(strDescription).toLowerCase();
            if(descLower.indexOf("hair") >= 0)
            {
               kind = "hair";
            }
            else if(descLower.indexOf("weapon") >= 0)
            {
               kind = "weapon";
            }
            else if(descLower.indexOf("back") >= 0 || descLower.indexOf("cape") >= 0)
            {
               kind = "back";
            }
            else if(descLower.indexOf("helm") >= 0 || descLower.indexOf("head") >= 0)
            {
               kind = "helmet";
            }
            _root.dfpsSilentLoadFile(mcDestination,strTarget,strDescription,remoteEntry,kind);
            return;
         }
         return _root.dfpsOriginalConnLoadFile.call(_root.conn,mcDestination,strTarget,strDescription);
      };
   }

   // Defensive second layer: some class/item SWFs call game.mcConn.loadClip()
   // directly rather than conn.loadFile(). Remote destinations are routed to
   // the silent loader; local destinations still use the untouched stock loader.
   if(_root.game.mcConn != undefined && _root.game.mcConn.loadClip != undefined)
   {
      _root.dfpsOriginalMcConnLoadClip = _root.game.mcConn.loadClip;
      _root.game.mcConn.loadClip = function(mcDestination,strTarget,strDescription)
      {
         var remoteEntry = _root.dfpsFindRemoteEntryForClip(mcDestination);
         if(remoteEntry != undefined)
         {
            _root.dfpsSilentLoadAbsolute(mcDestination,String(strTarget),String(strDescription),remoteEntry,"equipment");
            return;
         }
         return _root.dfpsOriginalMcConnLoadClip.call(_root.game.mcConn,mcDestination,strTarget,strDescription);
      };
   }

   _root.dfpsApplyRemoteFacing = function(entry)
   {
      if(entry == undefined || entry.clip == undefined)
      {
         return;
      }

      var magnitude = Math.abs(Number(entry.scaleX));
      if(isNaN(magnitude) || magnitude == 0)
      {
         magnitude = Math.abs(Number(entry.clip._xscale));
      }
      if(isNaN(magnitude) || magnitude == 0)
      {
         magnitude = 100;
      }

      var direction = String(entry.dir).toUpperCase();
      var faceLeft = Number(entry.scaleX) < 0;
      if(direction == "W" || direction == "NW" || direction == "SW")
      {
         faceLeft = true;
      }
      else if(direction == "E" || direction == "NE" || direction == "SE")
      {
         faceLeft = false;
      }

      entry.clip._xscale = faceLeft ? - magnitude : magnitude;
      entry.clip.isRotated = faceLeft;
      entry.clip.moveDir = direction;
      if(!isNaN(Number(entry.scaleY)) && Number(entry.scaleY) != 0)
      {
         entry.clip._yscale = Number(entry.scaleY);
      }
   };

   // Position stays network/interpolation driven, so no remote walkTo() call can
   // execute collision, room triggers, NPC scripts or local-player callbacks.
   // Each remote class plays its own native timeline. A stopped Walk timeline is
   // restarted while movement is still active, and non-walk states follow the
   // sender's real class frame (Idle/Fight/Attack/Cast/etc.).
   _root.dfpsApplyRemoteAnimation = function(entry,force)
   {
      if(entry == undefined || entry.clip == undefined || entry.appearanceReady != true)
      {
         return;
      }

      var now = getTimer();
      var remainingX = Number(entry.targetX) - Number(entry.currentX);
      var remainingY = Number(entry.targetY) - Number(entry.currentY);
      var remainingSq = remainingX * remainingX + remainingY * remainingY;
      var shouldWalk = entry.moving == true || remainingSq > 0.04 || now < Number(entry.movingUntil);

      if(shouldWalk)
      {
         // DragonFable class timelines use onMove at the final Walk frame to
         // jump straight back to the class-specific Walk label. Setting this
         // flag gives remote avatars the same seamless native loop as the local
         // player. No walkTo/onEnterWalk is installed, so collision, triggers
         // and local-player movement remain completely untouched.
         entry.clip.onMove = true;

         if(force == true || entry.renderAnimationMode != "walk")
         {
            entry.renderAnimationMode = "walk";
            entry.renderAnimationKey = "Walk";
            entry.clip.gotoAndPlay("Walk");
            entry.lastObservedAnimFrame = Number(entry.clip._currentframe);
            entry.lastAnimationAdvanceAt = now;
            return;
         }

         var walkFrame = Number(entry.clip._currentframe);
         if(walkFrame != Number(entry.lastObservedAnimFrame))
         {
            entry.lastObservedAnimFrame = walkFrame;
            entry.lastAnimationAdvanceAt = now;
         }
         else if(now - Number(entry.lastAnimationAdvanceAt) > 250)
         {
            // Fallback only for unusual classes that ignore onMove. Normal
            // classes loop natively and never enter this recovery path.
            entry.clip.gotoAndPlay("Walk");
            entry.lastObservedAnimFrame = Number(entry.clip._currentframe);
            entry.lastAnimationAdvanceAt = now;
         }
         return;
      }

      entry.clip.onMove = false;
      var animationKey = String(entry.animSerial) + "|" + String(entry.animation);
      if(force == true || entry.renderAnimationMode != "state" || entry.renderAnimationKey != animationKey)
      {
         entry.renderAnimationMode = "state";
         entry.renderAnimationKey = animationKey;
         entry.lastFrameSyncAt = now;

         var desiredFrame = Number(entry.frame);
         if(!isNaN(desiredFrame) && desiredFrame > 0)
         {
            // The sender and receiver use the same runtime class SWF, so its
            // absolute frame is the most accurate class-specific state signal.
            entry.clip.gotoAndPlay(desiredFrame);
         }
         else if(String(entry.animation) != "" && String(entry.animation) != "undefined" && String(entry.animation) != "null")
         {
            entry.clip.gotoAndPlay(String(entry.animation));
         }
         else
         {
            entry.clip.gotoAndPlay("Idle");
         }
      }
      else if(now - Number(entry.lastFrameSyncAt) > 350)
      {
         var incomingFrame = Number(entry.frame);
         var renderedFrame = Number(entry.clip._currentframe);
         if(!isNaN(incomingFrame) && incomingFrame > 0 && Math.abs(incomingFrame - renderedFrame) > 18)
         {
            entry.clip.gotoAndPlay(incomingFrame);
            entry.lastFrameSyncAt = now;
         }
      }
   };

   _root.dfpsDestroyRemotePlayer = function(entry)
   {
      if(entry == undefined)
      {
         return;
      }

      entry.loaderGeneration = Number(entry.loaderGeneration) + 1;

      if(entry.pendingLoads != undefined)
      {
         for(var p = entry.pendingLoads.length - 1; p >= 0; p--)
         {
            var loadInfo = entry.pendingLoads[p];
            if(loadInfo != undefined && loadInfo.loader != undefined && loadInfo.target != undefined)
            {
               loadInfo.loader.unloadClip(loadInfo.target);
            }
         }
         entry.pendingLoads = new Array();
      }

      if(entry.unitIndex != undefined && entry.unitIndex >= 0 && _root.game.Units != undefined && _root.game.Units.arrUnits != undefined)
      {
         if(_root.game.Units.arrUnits[entry.unitIndex] == entry.avatar)
         {
            delete _root.game.Units.arrUnits[entry.unitIndex];
         }
      }

      if(entry.avatar != undefined)
      {
         entry.avatar.destroy();
      }
      if(entry.clip != undefined)
      {
         entry.clip.removeMovieClip();
      }
      if(entry.statusClip != undefined)
      {
         entry.statusClip.removeMovieClip();
      }
      if(entry.label != undefined)
      {
         entry.label.removeTextField();
      }

      entry.avatar = undefined;
      entry.clip = undefined;
      entry.statusClip = undefined;
      entry.label = undefined;
      entry.parent = undefined;
      entry.unitIndex = -1;
      entry.appearanceReady = false;
      entry.appearanceLoading = false;
      entry.renderAnimationMode = undefined;
      entry.renderAnimationKey = undefined;
      entry.appliedEquipmentKey = "";
   };

   _root.dfpsClearRemotePlayers = function()
   {
      for(var remoteKey in _root.dfpsRemotePlayers)
      {
         _root.dfpsDestroyRemotePlayer(_root.dfpsRemotePlayers[remoteKey]);
         delete _root.dfpsRemotePlayers[remoteKey];
      }
   };

   _root.dfpsMakeRuntimeEquipmentItem = function(spot,fileName,itemType,itemTypeDetail)
   {
      fileName = String(fileName);
      if(fileName == "" || fileName == "undefined" || fileName == "null" || fileName.toLowerCase() == "none")
      {
         return undefined;
      }

      var item = new Object();
      item.strEquipSpot = String(spot);
      item.strFileName = fileName;
      item.strItemType = String(itemType);
      item.strType = String(itemTypeDetail);
      item.dfpsEquipped = true;
      item.bitEquipped = 0;
      item.used = false;
      item.showed = false;
      item.slotted = false;
      return item;
   };

   _root.dfpsBuildRuntimeEquipmentItems = function(entry)
   {
      var items = new Array();
      if(entry == undefined || entry.equipmentKnown != true)
      {
         return items;
      }

      var item;
      if(entry.weaponVisible == true)
      {
         item = _root.dfpsMakeRuntimeEquipmentItem("Weapon",entry.weaponFile,entry.weaponItemType,entry.weaponType);
         if(item != undefined)
         {
            items.push(item);
         }
      }
      if(entry.backVisible == true)
      {
         item = _root.dfpsMakeRuntimeEquipmentItem("Back",entry.backFile,entry.backItemType,entry.backType);
         if(item != undefined)
         {
            items.push(item);
         }
      }
      if(entry.headVisible == true)
      {
         item = _root.dfpsMakeRuntimeEquipmentItem("Head",entry.headFile,entry.headItemType,entry.headType);
         if(item != undefined)
         {
            items.push(item);
         }
      }
      return items;
   };

   _root.dfpsApplyRuntimeEquipment = function(entry,force)
   {
      if(entry == undefined || entry.avatar == undefined || entry.avatar.ptrObj == undefined || entry.avatar.ptrMC == undefined || entry.appearanceReady != true)
      {
         return;
      }
      if(entry.equipmentKnown != true)
      {
         if(force == true)
         {
            entry.avatar.equipSavedWeaponConfig();
         }
         return;
      }
      if(force != true && String(entry.appliedEquipmentKey) == String(entry.equipmentKey))
      {
         return;
      }

      var ptrMC = entry.avatar.ptrMC;

      // Cancel older remote-only equipment requests before reusing their target
      // clips. The class loader is never cancelled here.
      if(entry.pendingLoads != undefined)
      {
         for(var pendingIndex = entry.pendingLoads.length - 1; pendingIndex >= 0; pendingIndex--)
         {
            var pendingInfo = entry.pendingLoads[pendingIndex];
            if(pendingInfo != undefined && (pendingInfo.kind == "weapon" || pendingInfo.kind == "back" || pendingInfo.kind == "helmet"))
            {
               pendingInfo.finished = true;
               if(pendingInfo.loader != undefined && pendingInfo.target != undefined)
               {
                  pendingInfo.loader.unloadClip(pendingInfo.target);
               }
               entry.pendingLoads.splice(pendingIndex,1);
            }
         }
      }

      // Clear only this remote MovieClip. No local inventory, HUD, stats or
      // game.player state is touched.
      if(ptrMC.unEquipItem != undefined)
      {
         ptrMC.unEquipItem("Weapon");
      }
      if(ptrMC.weapon != undefined)
      {
         ptrMC.weapon._visible = false;
      }
      if(ptrMC.weapon2 != undefined)
      {
         ptrMC.weapon2._visible = false;
      }
      if(ptrMC.chest != undefined && ptrMC.chest.back != undefined)
      {
         ptrMC.chest.back._visible = false;
      }
      if(ptrMC.head != undefined)
      {
         if(ptrMC.head.helm != undefined)
         {
            ptrMC.head.helm._visible = false;
         }
         if(ptrMC.head.hair != undefined)
         {
            ptrMC.head.hair._visible = true;
         }
      }
      entry.avatar.bitHelmEquipped = false;

      entry.avatar.ptrObj.items = _root.dfpsBuildRuntimeEquipmentItems(entry);
      entry.avatar.equipSavedWeaponConfig();
      entry.appliedEquipmentKey = String(entry.equipmentKey);
   };

   _root.dfpsInstallSafeRemoteEquipment = function(remoteAvatar,entry)
   {
      remoteAvatar.dfpsEntry = entry;

      // If anything in DragonFable tries to reload this remote character later,
      // keep it on the silent loader instead of game.mcConn.
      remoteAvatar.loadCharacter = function()
      {
         return _root.dfpsSilentLoadRemoteCharacter(this.dfpsEntry);
      };

      remoteAvatar.equipSavedWeaponConfig = function()
      {
         if(this.ptrObj == undefined || this.ptrObj.items == undefined || this.ptrMC == undefined)
         {
            return;
         }

         for(var i = 0; i < this.ptrObj.items.length; i++)
         {
            var item = this.ptrObj.items[i];
            if(item == undefined || item.dfpsEquipped != true)
            {
               continue;
            }

            var spot = String(item.strEquipSpot).toLowerCase();
            var fileName = String(item.strFileName);

            if(spot == "weapon")
            {
               // Use the armor class's own visual weapon method. Any file loads
               // it starts are intercepted by the remote-safe wrappers above.
               if(this.ptrMC.equipItem != undefined)
               {
                  this.ptrMC.equipItem(item.strEquipSpot,item.strItemType,fileName);
                  if(this.ptrMC.weapon != undefined)
                  {
                     this.ptrMC.weapon._visible = true;
                  }
               }
            }
            else if(spot == "back" || spot == "cape")
            {
               if(this.ptrMC.chest != undefined && this.ptrMC.chest.back != undefined)
               {
                  _root.dfpsSilentLoadFile(this.ptrMC.chest.back,fileName,"DFPS Remote Back",this.dfpsEntry,"back");
                  this.ptrMC.chest.back._visible = true;
               }
            }
            else if(spot == "head" || spot == "helm")
            {
               if(this.ptrMC.head != undefined && this.ptrMC.head.helm != undefined)
               {
                  _root.dfpsSilentLoadFile(this.ptrMC.head.helm,fileName,"DFPS Remote Helmet",this.dfpsEntry,"helmet");
                  this.ptrMC.head.helm._visible = true;
                  this.bitHelmEquipped = true;

                  if(this.ptrMC.head.hair != undefined)
                  {
                     if(String(item.strType).toLowerCase() == "pierce")
                     {
                        this.ptrMC.head.hair._visible = true;
                     }
                     else
                     {
                        this.ptrMC.head.hair._visible = false;
                     }
                  }
               }
            }

            item.used = true;
            item.showed = true;
         }
      };

      remoteAvatar.loadSavedWeaponConfig = function()
      {
         this.equipSavedWeaponConfig();
      };

      // Remote units do not own the local HUD.
      remoteAvatar.refreshStatusBar = function()
      {
      };
   };

   _root.dfpsBuildRemoteAvatar = function(entry,appearanceXml,appearanceVersion)
   {
      var localAvatar = _root.game.avatar.ptrMC;
      if(localAvatar == undefined || localAvatar._parent == undefined || _root.game.Units == undefined || _root.game.stage.map == undefined)
      {
         return "deferred";
      }

      // Never start a remote appearance build while DragonFable is loading the
      // local character/map. JS will retry the same appearance shortly.
      if(_root.game.mcConn != undefined)
      {
         if(_root.game.mcConn._visible == true || Number(_root.game.mcConn.intLoadCount) > 0)
         {
            return "deferred";
         }
      }

      var appearanceObj = _root.dfpsParseAppearance(appearanceXml);
      if(appearanceObj == undefined)
      {
         return "failed";
      }

      // PvpProjection can contain the saved/default class. Multiplayer must
      // render the class currently loaded on the remote player's client.
      if(Number(entry.classId) > 0)
      {
         appearanceObj.ClassID = Number(entry.classId);
      }
      if(entry.classFile != undefined && String(entry.classFile) != "" && String(entry.classFile) != "undefined" && String(entry.classFile) != "null")
      {
         appearanceObj.strClassFileName = String(entry.classFile);
      }
      if(entry.equipmentKnown == true)
      {
         appearanceObj.items = _root.dfpsBuildRuntimeEquipmentItems(entry);
      }

      var oldTargetX = entry.targetX;
      var oldTargetY = entry.targetY;
      var oldCurrentX = entry.currentX;
      var oldCurrentY = entry.currentY;
      var oldScaleX = entry.scaleX;
      var oldScaleY = entry.scaleY;
      var oldDisplayName = entry.displayName;
      var oldClassId = entry.classId;
      var oldClassFile = entry.classFile;
      var oldMoving = entry.moving;
      var oldFailureCount = Number(entry.appearanceFailureCount);

      _root.dfpsDestroyRemotePlayer(entry);

      entry.targetX = oldTargetX;
      entry.targetY = oldTargetY;
      entry.currentX = oldCurrentX;
      entry.currentY = oldCurrentY;
      entry.scaleX = oldScaleX;
      entry.scaleY = oldScaleY;
      entry.displayName = oldDisplayName;
      entry.classId = oldClassId;
      entry.classFile = oldClassFile;
      entry.moving = oldMoving;
      entry.renderAnimationMode = undefined;
      entry.renderAnimationKey = undefined;
      entry.appliedEquipmentKey = "";
      entry.appearanceVersion = String(appearanceVersion);
      entry.appearanceObject = appearanceObj;
      entry.appearanceXml = String(appearanceXml);
      entry.appearanceFailureCount = oldFailureCount;
      entry.appearanceFailed = false;
      entry.appearanceLoading = true;
      entry.appearanceReady = false;
      entry.pendingLoads = new Array();

      var parent = localAvatar._parent;
      var key = "dfpsRemoteAvatar_" + entry.remoteId;
      var statusKey = key + "_status";

      parent.createEmptyMovieClip(key,parent.getNextHighestDepth());
      var remoteClip = parent[key];
      if(remoteClip == undefined)
      {
         entry.appearanceLoading = false;
         return "failed";
      }

      parent.createEmptyMovieClip(statusKey,parent.getNextHighestDepth());
      var statusClip = parent[statusKey];
      if(statusClip != undefined)
      {
         statusClip._visible = false;
         statusClip._x = -10000;
         statusClip._y = -10000;
      }

      var unitIndex = _root.dfpsAllocateUnitIndex();
      if(unitIndex < 0)
      {
         remoteClip.removeMovieClip();
         if(statusClip != undefined)
         {
            statusClip.removeMovieClip();
         }
         entry.appearanceLoading = false;
         return "failed";
      }

      var remoteAvatar = new Avatar("friend",appearanceObj);
      remoteAvatar.setPtrMC(remoteClip);
      remoteAvatar.setPtrSB(statusClip);
      _root.dfpsInstallSafeRemoteEquipment(remoteAvatar,entry);

      _root.game.Units.arrUnits[unitIndex] = remoteAvatar;

      var startPoint = _root.dfpsMapPointToParent(parent,entry.currentX,entry.currentY);
      remoteClip._x = startPoint.x;
      remoteClip._y = startPoint.y;
      remoteClip._alpha = 100;
      remoteClip._visible = false;

      entry.avatar = remoteAvatar;
      entry.clip = remoteClip;
      entry.statusClip = statusClip;
      entry.parent = parent;
      entry.unitIndex = unitIndex;
      entry.label = _root.dfpsCreateNameplate(parent,entry,key);

      if(_root.dfpsSilentLoadRemoteCharacter(entry) == false)
      {
         entry.appearanceLoading = false;
         entry.appearanceFailed = true;
         return "failed";
      }
      return "loading";
   };

   _root.dfpsWorldTick = function()
   {
      var avatar = _root.game.avatar.ptrMC;
      var playerObj = _root.game.player.ptrObj;
      var map = _root.game.stage.map;
      if(avatar == undefined || playerObj == undefined || map == undefined)
      {
         return;
      }

      var mapId = _root.dfpsGetMapId();
      if(mapId == "")
      {
         return;
      }

      if(_root.dfpsIsPrivateWorld())
      {
         if(_root.dfpsPrivateMode != true)
         {
            _root.dfpsPrivateMode = true;
            _root.dfpsClearRemotePlayers();
            _root.dfpsCurrentMapId = "";
            _root.dfpsLastLocalMapX = undefined;
            _root.dfpsLastLocalMapY = undefined;
            _root.dfpsLocalMovingUntil = 0;
         }

         flash.external.ExternalInterface.call("hiperesp.dfps.externalInterface.worldState",{
            privateMode:1
         });
         return;
      }

      if(_root.dfpsPrivateMode == true)
      {
         _root.dfpsPrivateMode = false;
         _root.dfpsCurrentMapId = "";
         _root.dfpsLastLocalMapX = undefined;
         _root.dfpsLastLocalMapY = undefined;
         _root.dfpsLocalMovingUntil = 0;
      }

      if(_root.dfpsCurrentMapId != mapId)
      {
         _root.dfpsClearRemotePlayers();
         _root.dfpsCurrentMapId = mapId;
         _root.dfpsLastLocalMapX = undefined;
         _root.dfpsLastLocalMapY = undefined;
         _root.dfpsLocalMovingUntil = 0;
         _root.dfpsLastLocalAnimFrame = undefined;
         _root.dfpsLastLocalAnimLabel = "";
         _root.dfpsLocalAnimSerial = _root.dfpsLocalAnimSerial + 1;
      }

      var mapPoint = _root.dfpsGetMapPosition(avatar);
      var now = getTimer();
      var movingSample = avatar.onMove == true;
      if(_root.dfpsLastLocalMapX != undefined && _root.dfpsLastLocalMapY != undefined)
      {
         var localDeltaX = Number(mapPoint.x) - Number(_root.dfpsLastLocalMapX);
         var localDeltaY = Number(mapPoint.y) - Number(_root.dfpsLastLocalMapY);
         if(localDeltaX * localDeltaX + localDeltaY * localDeltaY > 0.04)
         {
            movingSample = true;
         }
      }
      if(movingSample == true)
      {
         _root.dfpsLocalMovingUntil = now + 180;
      }
      var movingNow = movingSample == true || now < Number(_root.dfpsLocalMovingUntil);
      _root.dfpsLastLocalMapX = Number(mapPoint.x);
      _root.dfpsLastLocalMapY = Number(mapPoint.y);

      var activeClassFile = _root.dfpsGetActiveClassFile(avatar,playerObj);
      var animationLabel = _root.dfpsGetActiveAnimationLabel(avatar);
      var animationFrame = Number(avatar._currentframe);

      if(_root.dfpsLastLocalAnimFrame == undefined)
      {
         _root.dfpsLocalAnimSerial = _root.dfpsLocalAnimSerial + 1;
      }
      else
      {
         var frameDelta = animationFrame - Number(_root.dfpsLastLocalAnimFrame);
         if(animationLabel != _root.dfpsLastLocalAnimLabel || frameDelta < -2 || Math.abs(frameDelta) > 36)
         {
            _root.dfpsLocalAnimSerial = _root.dfpsLocalAnimSerial + 1;
         }
      }
      _root.dfpsLastLocalAnimFrame = animationFrame;
      _root.dfpsLastLocalAnimLabel = animationLabel;

      var runtimeEquipment = _root.dfpsGetRuntimeEquipment(playerObj,avatar);

      flash.external.ExternalInterface.call("hiperesp.dfps.externalInterface.worldState",{
         map:mapId,
         charId:Number(playerObj.CharID),
         classId:Number(playerObj.ClassID),
         classFile:activeClassFile,
         moving:movingNow ? 1 : 0,
         x:mapPoint.x,
         y:mapPoint.y,
         scaleX:avatar._xscale,
         scaleY:avatar._yscale,
         dir:avatar.moveDir,
         frame:animationFrame,
         animation:animationLabel,
         animSerial:_root.dfpsLocalAnimSerial,
         equipment:runtimeEquipment,
         privateMode:0,
         name:playerObj.strCharacterName
      });
   };

   _root.dfpsRemotePlayersBegin = function()
   {
      if(_root.dfpsIsPrivateWorld())
      {
         _root.dfpsClearRemotePlayers();
         return;
      }
      _root.dfpsRemoteGeneration = _root.dfpsRemoteGeneration + 1;
   };

   _root.dfpsRemotePlayer = function(remoteId,remoteName,remoteUsername,remoteCharId,remoteX,remoteY,remoteScaleX,remoteScaleY,remoteDir,remoteFrame,remoteClassId,remoteClassFile,remoteMoving,remoteAnimation,remoteAnimSerial,remoteWeaponFile,remoteWeaponItemType,remoteWeaponType,remoteWeaponVisible,remoteBackFile,remoteBackItemType,remoteBackType,remoteBackVisible,remoteHeadFile,remoteHeadItemType,remoteHeadType,remoteHeadVisible)
   {
      if(_root.dfpsIsPrivateWorld())
      {
         return;
      }

      var key = "remote_" + String(remoteId);
      var entry = _root.dfpsRemotePlayers[key];
      var now = getTimer();
      var newX = Number(remoteX);
      var newY = Number(remoteY);
      var incomingMoving = Number(remoteMoving) == 1 || String(remoteMoving) == "true";
      var isNew = entry == undefined;

      if(isNew)
      {
         var displayName = String(remoteName);
         if(displayName == "" || displayName == "undefined" || displayName == "null")
         {
            displayName = String(remoteUsername);
         }

         entry = new Object();
         entry.remoteId = String(remoteId);
         entry.charId = Number(remoteCharId);
         entry.classId = Number(remoteClassId);
         entry.classFile = String(remoteClassFile);
         entry.displayName = displayName;
         entry.currentX = newX;
         entry.currentY = newY;
         entry.targetX = newX;
         entry.targetY = newY;
         entry.previousX = newX;
         entry.previousY = newY;
         entry.previousAt = now;
         entry.latestX = newX;
         entry.latestY = newY;
         entry.latestAt = now;
         entry.velocityX = 0;
         entry.velocityY = 0;
         entry.lastPacketAt = now;
         entry.movingUntil = 0;
         entry.scaleX = Number(remoteScaleX);
         entry.scaleY = Number(remoteScaleY);
         entry.dir = String(remoteDir);
         entry.frame = Number(remoteFrame);
         entry.animation = String(remoteAnimation);
         entry.animSerial = Number(remoteAnimSerial);
         entry.renderAnimationMode = undefined;
         entry.renderAnimationKey = undefined;
         entry.generation = _root.dfpsRemoteGeneration;
         entry.appearanceVersion = "";
         entry.appearanceXml = "";
         entry.appearanceReady = false;
         entry.appearanceLoading = false;
         entry.appearanceFailed = false;
         entry.appearanceFailureCount = 0;
         entry.unitIndex = -1;
         entry.loaderGeneration = 1;
         entry.pendingLoads = new Array();
         entry.equipmentKnown = false;
         entry.equipmentKey = "";
         entry.appliedEquipmentKey = "";
         _root.dfpsRemotePlayers[key] = entry;
      }
      else
      {
         var lastSampleX = Number(entry.latestX);
         var lastSampleY = Number(entry.latestY);
         if(isNaN(lastSampleX))
         {
            lastSampleX = Number(entry.targetX);
         }
         if(isNaN(lastSampleY))
         {
            lastSampleY = Number(entry.targetY);
         }

         var sampleDeltaX = newX - lastSampleX;
         var sampleDeltaY = newY - lastSampleY;
         var movedInSample = sampleDeltaX * sampleDeltaX + sampleDeltaY * sampleDeltaY > 0.01;
         var snapDeltaX = newX - Number(entry.currentX);
         var snapDeltaY = newY - Number(entry.currentY);
         if(snapDeltaX * snapDeltaX + snapDeltaY * snapDeltaY > 22500)
         {
            // Teleports and room spawn changes must not glide across the map.
            entry.currentX = newX;
            entry.currentY = newY;
            entry.previousX = newX;
            entry.previousY = newY;
            entry.previousAt = now;
            entry.latestX = newX;
            entry.latestY = newY;
            entry.latestAt = now;
            entry.velocityX = 0;
            entry.velocityY = 0;
         }
         else if(movedInSample)
         {
            var sampleInterval = now - Number(entry.latestAt);
            if(isNaN(sampleInterval) || sampleInterval < 35)
            {
               sampleInterval = 35;
            }
            if(sampleInterval > 220)
            {
               sampleInterval = 220;
            }

            // Keep two authoritative snapshots. Rendering happens slightly
            // behind them, so packet jitter never becomes a visible stop.
            entry.previousX = lastSampleX;
            entry.previousY = lastSampleY;
            entry.previousAt = now - sampleInterval;
            entry.latestX = newX;
            entry.latestY = newY;
            entry.latestAt = now;

            var rawVelocityX = sampleDeltaX / sampleInterval;
            var rawVelocityY = sampleDeltaY / sampleInterval;
            entry.velocityX = Number(entry.velocityX) * 0.25 + rawVelocityX * 0.75;
            entry.velocityY = Number(entry.velocityY) * 0.25 + rawVelocityY * 0.75;
         }
         else if(incomingMoving != true)
         {
            // Refresh an already-settled snapshot without collapsing a final
            // buffered movement segment before it reaches the real endpoint.
            var settleX = Number(entry.latestX) - Number(entry.currentX);
            var settleY = Number(entry.latestY) - Number(entry.currentY);
            if(settleX * settleX + settleY * settleY < 0.09)
            {
               entry.previousX = Number(entry.latestX);
               entry.previousY = Number(entry.latestY);
               entry.previousAt = now - 1;
               entry.latestAt = now;
               entry.velocityX = 0;
               entry.velocityY = 0;
            }
         }
         entry.lastPacketAt = now;
      }

      entry.generation = _root.dfpsRemoteGeneration;
      entry.charId = Number(remoteCharId);

      var incomingClassId = Number(remoteClassId);
      var incomingClassFile = String(remoteClassFile);
      var runtimeClassChanged = false;
      if(entry.classFile != undefined && String(entry.classFile) != "" && incomingClassFile != "" && String(entry.classFile) != incomingClassFile)
      {
         runtimeClassChanged = true;
      }
      else if(Number(entry.classId) > 0 && incomingClassId > 0 && Number(entry.classId) != incomingClassId)
      {
         runtimeClassChanged = true;
      }

      entry.classId = incomingClassId;
      entry.classFile = incomingClassFile;

      var oldTargetX = Number(entry.targetX);
      var oldTargetY = Number(entry.targetY);
      var targetDeltaX = newX - oldTargetX;
      var targetDeltaY = newY - oldTargetY;
      var targetMoved = targetDeltaX * targetDeltaX + targetDeltaY * targetDeltaY > 0.01;
      entry.moving = incomingMoving;
      if(entry.moving == true || targetMoved)
      {
         entry.movingUntil = now + 240;
      }

      if(runtimeClassChanged)
      {
         // Cached XML can be reused, but force DFPS_RemoteAppearance to rebuild
         // it with the newly loaded runtime class file.
         entry.appearanceVersion = "";
      }

      entry.targetX = newX;
      entry.targetY = newY;
      entry.scaleX = Number(remoteScaleX);
      entry.scaleY = Number(remoteScaleY);
      entry.dir = String(remoteDir);
      entry.frame = Number(remoteFrame);
      entry.animation = String(remoteAnimation);
      entry.animSerial = Number(remoteAnimSerial);

      var equipmentKey = String(remoteWeaponFile) + "|" + String(remoteWeaponItemType) + "|" + String(remoteWeaponType) + "|" + String(remoteWeaponVisible) + "|" +
                         String(remoteBackFile) + "|" + String(remoteBackItemType) + "|" + String(remoteBackType) + "|" + String(remoteBackVisible) + "|" +
                         String(remoteHeadFile) + "|" + String(remoteHeadItemType) + "|" + String(remoteHeadType) + "|" + String(remoteHeadVisible);
      var equipmentChanged = entry.equipmentKnown == true && String(entry.equipmentKey) != equipmentKey;
      entry.weaponFile = String(remoteWeaponFile);
      entry.weaponItemType = String(remoteWeaponItemType);
      entry.weaponType = String(remoteWeaponType);
      entry.weaponVisible = Number(remoteWeaponVisible) == 1 || String(remoteWeaponVisible) == "true";
      entry.backFile = String(remoteBackFile);
      entry.backItemType = String(remoteBackItemType);
      entry.backType = String(remoteBackType);
      entry.backVisible = Number(remoteBackVisible) == 1 || String(remoteBackVisible) == "true";
      entry.headFile = String(remoteHeadFile);
      entry.headItemType = String(remoteHeadItemType);
      entry.headType = String(remoteHeadType);
      entry.headVisible = Number(remoteHeadVisible) == 1 || String(remoteHeadVisible) == "true";
      entry.equipmentKnown = true;
      entry.equipmentKey = equipmentKey;

      if(equipmentChanged && runtimeClassChanged != true)
      {
         _root.dfpsApplyRuntimeEquipment(entry,true);
      }

      var newDisplayName = String(remoteName);
      if(newDisplayName == "" || newDisplayName == "undefined" || newDisplayName == "null")
      {
         newDisplayName = String(remoteUsername);
      }
      entry.displayName = newDisplayName;

      if(entry.label != undefined)
      {
         entry.label.text = entry.displayName;
      }
   };

   _root.dfpsRemoteAppearance = function(remoteId,remoteCharId,appearanceVersion,appearanceXml)
   {
      if(_root.dfpsIsPrivateWorld())
      {
         return "deferred";
      }

      var key = "remote_" + String(remoteId);
      var entry = _root.dfpsRemotePlayers[key];
      if(entry == undefined)
      {
         return "deferred";
      }
      if(Number(entry.charId) != Number(remoteCharId))
      {
         return "failed";
      }

      var requestedVersion = String(appearanceVersion);
      if(entry.appearanceVersion == requestedVersion)
      {
         if(entry.appearanceReady == true)
         {
            return "ready";
         }
         if(entry.appearanceLoading == true)
         {
            return "loading";
         }

         // Retry a failed class load a limited number of times. Equipment
         // failures do not mark the whole appearance as failed.
         if(entry.appearanceFailed == true && Number(entry.appearanceFailureCount) >= 3)
         {
            return "failed";
         }
      }

      return _root.dfpsBuildRemoteAvatar(entry,String(appearanceXml),requestedVersion);
   };

   _root.dfpsRemotePlayersEnd = function()
   {
      if(_root.dfpsIsPrivateWorld())
      {
         _root.dfpsClearRemotePlayers();
         return;
      }

      for(var remoteKey in _root.dfpsRemotePlayers)
      {
         var entry = _root.dfpsRemotePlayers[remoteKey];
         if(entry == undefined || entry.generation != _root.dfpsRemoteGeneration)
         {
            _root.dfpsDestroyRemotePlayer(entry);
            delete _root.dfpsRemotePlayers[remoteKey];
         }
      }
   };

   _root.dfpsSmoothRemotePlayers = function()
   {
      if(_root.dfpsIsPrivateWorld())
      {
         _root.dfpsClearRemotePlayers();
         return;
      }

      var localAvatar = _root.game.avatar.ptrMC;
      var map = _root.game.stage.map;
      if(localAvatar == undefined || map == undefined)
      {
         return;
      }

      var now = getTimer();
      for(var remoteKey in _root.dfpsRemotePlayers)
      {
         var entry = _root.dfpsRemotePlayers[remoteKey];
         if(entry == undefined)
         {
            continue;
         }

         var previousAt = Number(entry.previousAt);
         var latestAt = Number(entry.latestAt);
         var renderAt = now - Number(_root.dfpsInterpolationDelay);
         if(isNaN(previousAt) || isNaN(latestAt) || latestAt <= previousAt)
         {
            entry.currentX = Number(entry.latestX);
            entry.currentY = Number(entry.latestY);
         }
         else if(renderAt <= previousAt)
         {
            entry.currentX = Number(entry.previousX);
            entry.currentY = Number(entry.previousY);
         }
         else if(renderAt < latestAt)
         {
            var progress = (renderAt - previousAt) / (latestAt - previousAt);
            entry.currentX = Number(entry.previousX) + (Number(entry.latestX) - Number(entry.previousX)) * progress;
            entry.currentY = Number(entry.previousY) + (Number(entry.latestY) - Number(entry.previousY)) * progress;
         }
         else
         {
            var extrapolation = renderAt - latestAt;
            if(extrapolation < 0)
            {
               extrapolation = 0;
            }
            if(extrapolation > Number(_root.dfpsMaxExtrapolation))
            {
               extrapolation = Number(_root.dfpsMaxExtrapolation);
            }

            if(entry.moving == true)
            {
               entry.currentX = Number(entry.latestX) + Number(entry.velocityX) * extrapolation;
               entry.currentY = Number(entry.latestY) + Number(entry.velocityY) * extrapolation;
            }
            else
            {
               entry.currentX = Number(entry.latestX);
               entry.currentY = Number(entry.latestY);
               entry.velocityX = 0;
               entry.velocityY = 0;
            }
         }

         if(entry.clip != undefined && entry.parent != undefined)
         {
            if(entry.parent != localAvatar._parent)
            {
               // The avatar layer was replaced by a room transition. Destroy the
               // old instance; JS will re-apply the cached appearance on the new
               // active layer without touching the global game loader.
               _root.dfpsDestroyRemotePlayer(entry);
               continue;
            }

            var point = _root.dfpsMapPointToParent(entry.parent,entry.currentX,entry.currentY);
            entry.clip._x = point.x;
            entry.clip._y = point.y;

            _root.dfpsApplyRemoteFacing(entry);
            _root.dfpsApplyRemoteAnimation(entry,false);
         }

         if(entry.label != undefined && entry.parent != undefined)
         {
            var labelPoint = _root.dfpsMapPointToParent(entry.parent,entry.currentX,entry.currentY);
            entry.label._x = labelPoint.x - entry.label._width / 2;
            entry.label._y = labelPoint.y - 115;
         }
      }
   };

   flash.external.ExternalInterface.addCallback("DFPS_RemotePlayersBegin",_root,_root.dfpsRemotePlayersBegin);
   flash.external.ExternalInterface.addCallback("DFPS_RemotePlayer",_root,_root.dfpsRemotePlayer);
   flash.external.ExternalInterface.addCallback("DFPS_RemoteAppearance",_root,_root.dfpsRemoteAppearance);
   flash.external.ExternalInterface.addCallback("DFPS_RemotePlayersEnd",_root,_root.dfpsRemotePlayersEnd);
   flash.external.ExternalInterface.addCallback("DFPS_RemotePlayersClear",_root,_root.dfpsClearRemotePlayers);

   _root.dfpsWorldTimer = setInterval(_root.dfpsWorldTick,60);
   _root.dfpsSmoothTimer = setInterval(_root.dfpsSmoothRemotePlayers,20);
   _root.dfpsRemoteLoaderTimer = setInterval(_root.dfpsSilentLoaderWatchdog,250);
}
